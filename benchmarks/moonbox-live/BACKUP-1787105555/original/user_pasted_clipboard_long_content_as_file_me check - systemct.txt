   * Native read/write/bash/fd/rg/ls and individual destructive tools are NOT

  export const MCPPROXY_GATEWAY_TOOLS = [

     "call_tool_read",

     "call_tool_destructive",

  export type AgentConfigShape = {

  };

  /** Resolve repo root by walking up from this file until bin/pi is found. */
  export function findRepoRoot(): string {
     let dir = __dirname;
     for (let i = 0; i < 10; i++) {
        if (existsSync(join(dir, "bin", "pi"))) return dir;
        const parent = dirname(dir);
        if (parent === dir) break;
        dir = parent;
     }
     throw new Error("Could not locate repo root (bin/pi not found walking up from test dir)");
  }

  /** Parse the live deployed agent config (~/.pi/agent/config.yaml). */
  export function loadLiveAgentConfig(): AgentConfigShape | null {
     const path = join(homedir(), ".pi", "agent", "config.yaml");
     if (!existsSync(path)) return null;
     try {
        return parseYaml(readFileSync(path, "utf-8")) as AgentConfigShape;
     } catch {
        return null;
     }
  }

  /** Tools present in allowed-tools that are NOT part of the gateway family. */
  export function extraNonGatewayTools(allowedTools: string[] | undefined): string[] {
     return (allowedTools ?? []).filter((t) => !MCPPROXY_GATEWAY_TOOLS.includes(t as (typeof MCPPROXY_GATEWAY_TOOLS)[number]));
  }

  /** True iff allowed-tools is non-empty and a strict subset of the gateway family. */
  export function isGatewayOnly(allowedTools: string[] | undefined): boolean {
     const tools = allowedTools ?? [];
     return tools.length > 0 && tools.every((t) => MCPPROXY_GATEWAY_TOOLS.includes(t as (typeof MCPPROXY_GATEWAY_TOOLS)[number]));
  }

  /** True iff broadcaster is NOT present in the subagent whitelist. */
