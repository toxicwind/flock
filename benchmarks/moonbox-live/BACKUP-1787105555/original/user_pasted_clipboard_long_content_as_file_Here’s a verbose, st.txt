# ─── STATE MANAGER ──────────────────────────────────────────────────
class State:
    def __init__(self):
        self.data = {}
        self.load()
    def load(self):
        if os.path.exists(STATE_FILE):
            with open(STATE_FILE) as f:
                self.data = json.load(f)
        else:
            self.data = {"done": []}
    def save(self):
        with open(STATE_FILE, 'w') as f:
            json.dump(self.data, f, indent=2)
    def is_done(self, task_name):
        return task_name in self.data.get("done", [])
    def mark_done(self, task_name):
        if task_name not in self.data.get("done", []):
            self.data.setdefault("done", []).append(task_name)
            self.save()

class StreamLogger:
    def __init__(self, name):
        self.path = f"{OUT_DIR}/{name}.jsonl"
        self.buf = []
        self.lock = asyncio.Lock()
    async def log(self, d):
        async with self.lock:
            self.buf.append(json.dumps(d, default=str))
            if len(self.buf) >= 10:
                await self.flush()

        if self.buf:
            async with aiofiles.open(self.path, 'a') as f:
                await f.write('\n'.join(self.buf) + '\n')
            self.buf.clear()
    async def close(self):
        await self.flush()

How to run (verbose & resumable)

```bash
# Standalone
python3 /tmp/ultimate_racer.py

# Inside ipython (with magic)
ipython -c "%run -i /tmp/ultimate_racer.py"
```

It prints every step, shows progress, and saves state to /tmp/racer_state.json.
If you Ctrl+C and re‑run, it skips completed tasks (network scan, DNS, K8s, CDP, bind brute, gRPC, and each HTTP fuzz endpoint).

All results are appended to JSONL files in /mnt/agents/output/recon_stream/ – you can tail -f them while it runs.

If you want to force a re‑scan, just delete the state file: rm /tmp/racer_state.json.
