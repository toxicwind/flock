def test_endpoint(svc, method, params, auth_type):
    if auth_type == "jwt":
        headers = {"Authorization": f"Bearer {JWT}", "Content-Type": "application/json", "connect-protocol-version": "1"}
    elif auth_type == "bot":
        headers = {"Authorization": f"Bearer {NEW_API_KEY}", "Content-Type": "application/json", "connect-protocol-version": "1"}
    elif auth_type == "none":
        headers = {"Content-Type": "application/json", "connect-protocol-version": "1"}
    
    url = f"{BASE}/{svc}/{method}"
    try:
        r = requests.post(url, headers=headers, json=params, timeout=3)
        return (auth_type, svc, method, r.status_code, r.text[:150])
    except Exception as e:
        return (auth_type, svc, method, -1, str(e)[:50])

# Run all tests in parallel
all_tests = []
for svc, method, params in endpoints:
    for auth in ["jwt", "bot", "none"]:
        all_tests.append((svc, method, params, auth))

print(f"Running {len(all_tests)} tests in parallel...")
results = []
with concurrent.futures.ThreadPoolExecutor(max_workers=30) as executor:
    futures = {executor.submit(test_endpoint, svc, method, params, auth): (svc, method, auth) for svc, method, params, auth in all_tests}
    for future in concurrent.futures.as_completed(futures):
        result = future.result()
        results.append(result)
        auth_type, svc, method, status, body = result
        if status != 404 and status != 401 and status != 200:
            print(f"UNUSUAL [{auth_type}] {svc}/{method}: {status} - {body}")
