# DSSHRC v3.0 - Auto-chunk, auto-detect truncation, auto-continue

# Auto-chunk function - splits output to avoid truncation
chunk() {
    local data="$1"
    local prefix="${2:-chunk}"
    local maxlen=7000
    local total=${#data}
    local offset=0
    local chunk_num=0
    
    while [ $offset -lt $total ]; do
        local chunk_data="${data:$offset:$maxlen}"
        echo "=== CHUNK $prefix [$offset/$total] ==="
        echo "$chunk_data"
        offset=$((offset + maxlen))
        chunk_num=$((chunk_num + 1))
    done
    echo "=== TOTAL CHUNKS: $chunk_num ==="
}

# Smart execute with auto-log and auto-chunk
smart() {
    local cmd="$1"
    local logfile="/mnt/agents/.audit_logs/smart_$(date +%s)_$$.log"
    mkdir -p /mnt/agents/.audit_logs
    
    echo "[$(date -Iseconds)] EXEC: $cmd" >> "$logfile"
    local output
    output=$(bash -c "$cmd" 2>&1)
    local rc=$?
    
    echo "$output" >> "$logfile"
    echo "[$(date -Iseconds)] RC: $rc" >> "$logfile"
    
    # Auto-chunk if too long (stay under 10k limit)
    local len=${#output}
    if [ $len -gt 7000 ]; then
        chunk "$output" "smart_output"
    else
        echo "$output"
    fi
    
    return $rc
}

# Continue from offset in file
continue_at() {
    local file="$1"
    local offset="${2:-0}"
    local bytes="${3:-7000}"
    
    if [ -f "$file" ]; then
        dd if="$file" bs=1 skip="$offset" count="$bytes" 2>/dev/null
    else
        echo "File not found: $file"
    fi
}
