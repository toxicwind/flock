The messy output is a classic case of strings | grep on a binary: you get a firehose of noise, false positives, and garbage substrings. We need to extract meaningful secrets, not random bits.

New Clean Extraction Script

```python
#!/usr/bin/env python3
"""
envd_string_extractor.py – Intelligent, structured extraction from envd binary.
Filters noise, deduplicates, categorizes, and outputs clean JSON.
"""

import os
import sys
import re
import json
import subprocess
from collections import defaultdict
from typing import Dict, List, Set, Optional

# --- Configuration ---
BINARY_PATH = "/usr/local/bin/envd"
OUTPUT_FILE = "/tmp/envd_strings_clean.json"

# --- Patterns (tuned for meaningful content) ---
PATTERNS = {
    "url": re.compile(rb"https?://[a-zA-Z0-9./?=_-]+"),
    "jwt": re.compile(rb"eyJ[a-zA-Z0-9_-]+\.[a-zA-Z0-9_-]+\.[a-zA-Z0-9_-]+"),
    "api_key": re.compile(rb"km_[a-zA-Z0-9_]+|sk-[a-zA-Z0-9_]+"),
    "base64_key": re.compile(rb"^[A-Za-z0-9+/=]{32,64}$"),  # typical key length
    "canary": re.compile(rb"canary\s*[:=]\s*[0-9]+", re.I),
    "bind_token": re.compile(rb"bind[_-]?token\s*[:=]\s*[a-zA-Z0-9_\-]+", re.I),
    "secret": re.compile(rb"secret\s*[:=]\s*[a-zA-Z0-9_\-]+", re.I),
    "endpoint": re.compile(rb"endpoint\s*[:=]\s*[a-zA-Z0-9./:_\-]+", re.I),
}

# --- Noise filter (strings we want to ignore) ---
NOISE = {
    "keyFuncs", "keyKind", "keySize", "subkeys", "keyShares", "keyWiretag",
    "go/token", "tokenError", "tokenState", "tokenStack", "sortkey",
    "Rekey", "keyType", "keyConv", "keyZero", "keyLen", "Nsecret",
    "authorities", "binduEH", "9bindu", "expaL9", "H9", "L9P",
    "authn.key", "*api.API", "*api.Error", "*api.EnvVars", "*flate.token",
    "*token.Token", "*chi.endpoint", "unauthenH9", "ion/grpc",
}

def extract_strings(bin_path: str) -> List[str]:
    """Run strings on binary and return list of strings."""
    try:
        result = subprocess.run(
            ["strings", "-a", "-n", "4", bin_path],
            capture_output=True,
            check=True
        )
        return result.stdout.decode('utf-8', errors='ignore').splitlines()
    except Exception as e:
        print(f"Error running strings: {e}", file=sys.stderr)
        return []

def classify_string(s: str) -> Dict[str, Optional[str]]:
    """Classify a single string by pattern or mark as noise."""
    # Skip very short strings (likely noise)
    if len(s) < 6:
        return {"category": "noise", "value": s, "reason": "too short"}

    # Skip known noise
    if s in NOISE or s.lower() in NOISE:
        return {"category": "noise", "value": s, "reason": "known noise"}

    # Check patterns
    for cat, pat in PATTERNS.items():
        if pat.search(s.encode('utf-8', errors='ignore')):
            return {"category": cat, "value": s}

    # Check if it looks like a base64 string (length >= 32, only base64 chars)
    if re.fullmatch(r'^[A-Za-z0-9+/=]{32,}$', s):
        return {"category": "base64_candidate", "value": s}

    # If it contains colon or equals and looks like config (key=value)
    if re.search(r'[:=]', s) and len(s.split()) == 1:
        return {"category": "config", "value": s}

    # Otherwise, if it's a long alphanumeric, could be a key
    if len(s) >= 20 and re.fullmatch(r'^[A-Za-z0-9]+$', s):
        return {"category": "alphanumeric_key", "value": s}

    return {"category": "other", "value": s}

def main():
    if not os.path.exists(BINARY_PATH):
        print(f"Binary not found: {BINARY_PATH}", file=sys.stderr)
        sys.exit(1)

    raw = extract_strings(BINARY_PATH)
    print(f"Extracted {len(raw)} strings from {BINARY_PATH}")

    classified = []
    seen = set()
    for s in raw:
        s = s.strip()
        if not s or s in seen:
            continue
        seen.add(s)
        classified.append(classify_string(s))

    # Organize by category
    grouped = defaultdict(list)
    for item in classified:
        grouped[item["category"]].append(item["value"])

    # Remove duplicates within categories
    for cat in grouped:
        grouped[cat] = list(dict.fromkeys(grouped[cat]))

    # Write clean JSON output
    with open(OUTPUT_FILE, 'w') as f:
        json.dump({
            "total_strings": len(raw),
            "unique_strings": len(seen),
            "categories": dict(grouped)
        }, f, indent=2)

    # Also print a summary
    print("\n=== Envd Strings Extraction Summary ===")
    for cat, items in grouped.items():
        print(f"\n[{cat.upper()}] ({len(items)} items)")
        for item in items[:10]:  # show first 10
            print(f"  - {item[:100]}")
        if len(items) > 10:
            print(f"  ... and {len(items)-10} more")

    print(f"\nFull output saved to {OUTPUT_FILE}")

if __name__ == "__main__":
    main()
```
