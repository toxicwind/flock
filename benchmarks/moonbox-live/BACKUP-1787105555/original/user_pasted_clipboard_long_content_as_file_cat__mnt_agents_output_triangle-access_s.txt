"use client";

import { useState } from "react";

export default function QuoteCalculator() {
  const [pages, setPages] = useState(10);
  const [docType, setDocType] = useState("pdf");
  const [turnaround, setTurnaround] = useState("standard");
  const [compliance, setCompliance] = useState("wcag22aa");

  const baseRate = 8.0;
  const typeMultipliers: Record<string, number> = { pdf: 1.0, word: 1.2, powerpoint: 1.5, excel: 1.3, indesign: 1.8 };
  const urgencyMultipliers: Record<string, number> = { standard: 1.0, express: 1.5, rush: 2.0 };
  const complianceAddons: Record<string, number> = { wcag22aa: 0, wcag21aa: 0, section508: 0.1, pdfua: 0.15 };

  const volumeDiscount = pages >= 1000 ? 0.30 : pages >= 500 ? 0.20 : pages >= 100 ? 0.10 : 0;
  const baseCost = pages * baseRate * (typeMultipliers[docType] || 1);
  const total = baseCost * (urgencyMultipliers[turnaround] || 1) * (1 - volumeDiscount) * (1 + (complianceAddons[compliance] || 0));

  return (
    <div className="p-8 bg-[#111118] border border-white/5 rounded-2xl">
      <h2 className="text-2xl font-bold text-white mb-6">Calculate Your Quote</h2>
      <div className="space-y-6">
        <div>
          <label className="block text-sm font-medium text-gray-300 mb-2">Number of Pages: {pages}</label>
          <input
            type="range"
            min="1"
            max="1000"
            value={pages}
            onChange={(e) => setPages(parseInt(e.target.value))}
            className="w-full accent-[#00D4AA]"
          />
          <div className="flex justify-between text-xs text-gray-500 mt-1">
            <span>1 page</span>
            <span>1000 pages</span>
          </div>
        </div>

        <div>
          <label className="block text-sm font-medium text-gray-300 mb-2">Document Type</label>
          <div className="grid grid-cols-3 gap-2">
            {Object.entries({ pdf: "PDF", word: "Word", powerpoint: "PowerPoint", excel: "Excel", indesign: "InDesign" }).map(([key, label]) => (
              <button
                key={key}
                onClick={() => setDocType(key)}
                className={`px-4 py-2 rounded-lg text-sm font-medium transition ${
                  docType === key
                    ? "bg-[#00D4AA] text-[#0A0A0F]"
                    : "bg-[#1A1A24] text-gray-400 border border-white/10 hover:border-[#00D4AA]/30"
                }`}
              >
                {label}
              </button>
            ))}
          </div>
        </div>

        <div>
          <label className="block text-sm font-medium text-gray-300 mb-2">Turnaround</label>
          <div className="grid grid-cols-3 gap-2">
            {[
              { key: "standard", label: "5 days", multiplier: "1x" },
              { key: "express", label: "2 days", multiplier: "1.5x" },
              { key: "rush", label: "24h", multiplier: "2x" },
            ].map((opt) => (
              <button
                key={opt.key}
                onClick={() => setTurnaround(opt.key)}
                className={`px-4 py-2 rounded-lg text-sm font-medium transition ${
                  turnaround === opt.key
                    ? "bg-[#00D4AA] text-[#0A0A0F]"
                    : "bg-[#1A1A24] text-gray-400 border border-white/10 hover:border-[#00D4AA]/30"
                }`}
              >
                <div>{opt.label}</div>
                <div className="text-xs opacity-70">{opt.multiplier}</div>
              </button>
            ))}
          </div>
        </div>

        <div>
          <label className="block text-sm font-medium text-gray-300 mb-2">Compliance Level</label>
          <select
            value={compliance}
            onChange={(e) => setCompliance(e.target.value)}
            className="w-full px-4 py-3 bg-[#1A1A24] border border-white/10 rounded-xl text-white focus:outline-none focus:border-[#00D4AA]"
          >
            <option value="wcag22aa">WCAG 2.2 AA (Recommended)</option>
            <option value="wcag21aa">WCAG 2.1 AA</option>
            <option value="section508">Section 508</option>
            <option value="pdfua">PDF/UA</option>
          </select>
        </div>

        <div className="pt-6 border-t border-white/5">
          <div className="flex items-center justify-between mb-2">
            <span className="text-gray-400">Base Cost</span>
            <span className="text-white">${baseCost.toFixed(2)}</span>
          </div>
          {volumeDiscount > 0 && (
            <div className="flex items-center justify-between mb-2">
              <span className="text-[#00D4AA]">Volume Discount ({Math.round(volumeDiscount * 100)}%)</span>
              <span className="text-[#00D4AA]">-${(baseCost * volumeDiscount).toFixed(2)}</span>
            </div>
          )}
          <div className="flex items-center justify-between text-xl font-bold text-white pt-2">
            <span>Total</span>
            <span className="text-[#00D4AA]">${total.toFixed(2)}</span>
          </div>
          <p className="text-xs text-gray-500 mt-2">Guaranteed price. No hidden fees.</p>
        </div>
      </div>
    </div>
  );
}
