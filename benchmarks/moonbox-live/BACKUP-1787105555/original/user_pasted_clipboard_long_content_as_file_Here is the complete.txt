"""
morphe/context.py
Encapsulates runtime execution state, code transformations, and audit signals.
"""

from typing import Any, Dict, Optional


class MorpheHookContext:
    """
    Tracks state transitions during pre-lexical, AST-morphing, and post-execution stages.
    """
    def __init__(self, raw_source: str, filename: str = "<morphe-eval>", mode: str = "single"):
        self.raw_source: str = raw_source
        self.filename: str = filename
        self.mode: str = mode
        self.transformed_source: Optional[str] = None
        self.ast_mutated: bool = False
        self.short_circuit: bool = False
        self.execution_result: Any = None
        self.error: Optional[BaseException] = None
        self.elapsed_ms: float = 0.0
        self.metadata: Dict[str, Any] = {}

    def rewrite_source(self, new_source: str) -> None:
        """Mutate the raw lexical stream before AST generation."""
        self.transformed_source = new_source

    def abort_and_return(self, result: Any) -> None:
        """Short-circuit native evaluation and immediately return proxy value."""
        self.short_circuit = True
        self.execution_result = result

#### 4. `morphe/ast_morpher.py`

"""
morphe/ast_morpher.py
Dynamic AST transformers implementing NodeTraversers and on-the-fly syntax morphing.
"""

import ast
from typing import Optional


class MorpheASTTransformer(ast.NodeTransformer):
    """
    Intercepts and transforms AST nodes prior to bytecode compilation.
    """
    def __init__(self, enable_auto_audit: bool = True):
        super().__init__()
        self.enable_auto_audit = enable_auto_audit

    def visit_BinOp(self, node: ast.BinOp) -> ast.AST:
        """Intercept and log binary operations or rewrite arithmetic ops."""
        self.generic_visit(node)
        return node

    def visit_Assign(self, node: ast.Assign) -> ast.AST:
        """
        Intercepts assignment nodes. Wraps target variables or enriches
        metadata tracking.
        """
        self.generic_visit(node)
        return node

    def morph(self, tree: ast.AST) -> ast.AST:
        """Executes full-pass AST rewriting with missing location repair."""
        mutated_tree = self.visit(tree)
        ast.fix_missing_locations(mutated_tree)
        return mutated_tree

```

"""
morphe/shell_shims.py
Generates in-memory and bind-mountable executable proxies for Dash and Bash.
"""

import os
import stat
from pathlib import Path


def generate_dash_shim(target_dir: Path, real_dash_path: str = "/bin/dash") -> Path:
    """Generates an execution interceptor wrapper for /bin/dash."""
    shim_path = target_dir / "dash_shim"
    content = f"""#!/bin/sh
# --- Morphe Out-of-Band Dash Interceptor ---
echo "\\033[94m[Morphe Dash Intercept]\\033[0m Command: $*" >&2
exec {real_dash_path} "$@"
"""
    shim_path.write_text(content)
    shim_path.chmod(shim_path.stat().st_mode | stat.S_IEXEC | stat.S_IXGRP | stat.S_IXOTH)
    return shim_path


def generate_bashrc_shim(target_dir: Path) -> Path:
    """Generates global DEBUG-trap configuration for non-interactive and interactive Bash."""
    bashrc_path = target_dir / "morphe_bashrc"
    content = """# --- Morphe Out-of-Band Bash Interceptor ---
PS1="\\[\\033[36m\\][morphe-ns]\\[\\033[00m\\] \\u@\\h:\\w\\$ "
trap 'echo -e "\\033[33m[Morphe Bash Trace]\\033[0m Executing: $BASH_COMMAND" >&2' DEBUG
"""
    bashrc_path.write_text(content)
    return bashrc_path

```

#### 7. `morphe/kernel.py`

```python
"""
morphe/kernel.py
Provides the standalone MorpheAutoHookKernel and IPython singleton grafting routines.
"""

import ast
import code
import sys
import time
import types
from typing import Any, Callable, List, Optional
from morphe.ast_morpher import MorpheASTTransformer
from morphe.context import MorpheHookContext


class MorpheIPythonPatcher:
    """
    Live monkey-patcher for active IPython InteractiveShell instances.
    """
    _original_run_cell: Optional[Callable] = None
    _installed: bool = False

    @classmethod
    def install(cls) -> bool:
        """Grafts Morphe lifecycle hooks and AST transformers directly onto active IPython shell."""
        try:
            from IPython import get_ipython
            ip = get_ipython()
            if ip is None:
                from IPython.core.interactiveshell import InteractiveShell
                if InteractiveShell.initialized():
                    ip = InteractiveShell.instance()
            
            if ip is None:
                return False

            if cls._installed:
                return True

            # 1. Register AST Transformer
            transformer = MorpheASTTransformer()
            ip.ast_transformers.append(transformer)

            # 2. Register Native Event Hooks
            def _pre_run_cell(cell_info):
                print(f"\033[96m[Morphe IPython :: PRE]\033[0m Executing cell ({len(cell_info.raw_cell)} chars)")

            def _post_run_cell(result):
                status = "SUCCESS" if result.error_in_exec is None else f"ERROR: {result.error_in_exec}"
                print(f"\033[96m[Morphe IPython :: POST]\033[0m Execution finished -> {status}")

            ip.events.register("pre_run_cell", _pre_run_cell)
            ip.events.register("post_run_cell", _post_run_cell)

            cls._installed = True
            print("✓ Morphe auto-hooks grafted onto active IPython kernel.")
            return True
        except Exception as e:
            print(f"[Morphe Hook] Failed to graft IPython instance: {e}", file=sys.stderr)
            return False


class MorpheAutoHookKernel(code.InteractiveConsole):
    """
    Standalone interactive execution engine with dynamic AST morphing & lifecycle hooks.
    """
    def __init__(self, locals_dict: Optional[dict] = None, filename: str = "<morphe-kernel>"):
        super().__init__(locals=locals_dict, filename=filename)
        self.pre_run_hooks: List[Callable[[MorpheHookContext], None]] = []
        self.post_run_hooks: List[Callable[[MorpheHookContext], None]] = []
        self.ast_transformer = MorpheASTTransformer()

    def register_hook(self, event_type: str, callback: Callable) -> None:
        if event_type == "pre_run":
            self.pre_run_hooks.append(callback)
        elif event_type == "post_run":
            self.post_run_hooks.append(callback)

    def morph_and_compile(self, ctx: MorpheHookContext) -> Any:
        source = ctx.transformed_source if ctx.transformed_source is not None else ctx.raw_source
        parsed_ast = ast.parse(source, ctx.filename, ctx.mode)
        mutated_ast = self.ast_transformer.morph(parsed_ast)
        ctx.ast_mutated = True
        return compile(mutated_ast, ctx.filename, ctx.mode)

    def runsource(self, source: str, filename: str = "<input>", symbol: str = "single") -> bool:
        ctx = MorpheHookContext(raw_source=source, filename=filename, mode=symbol)

        # 1. Pre-execution Lifecycle Hooks
        for hook in self.pre_run_hooks:
            hook(ctx)

        if ctx.short_circuit:
            return False

        t0 = time.perf_counter()
        try:
            # 2. Syntax Morphing & Compilation
            code_obj = self.morph_and_compile(ctx)
            # 3. Native Execution
            ctx.execution_result = self.runcode(code_obj)
        except (OverflowError, SyntaxError, ValueError) as err:
            ctx.error = err
            self.showsyntaxerror(filename)
            return False
        except BaseException as err:
            ctx.error = err
            self.showtraceback()
        finally:
            ctx.elapsed_ms = (time.perf_counter() - t0) * 1000.0
            # 4. Post-execution Lifecycle Hooks
            for hook in self.post_run_hooks:
                hook(ctx)

        return False


def main():
    """Demo runner executing standalone kernel."""
    kernel = MorpheAutoHookKernel()
    
    def on_pre(ctx: MorpheHookContext):
        print(f"  [Morphe Standalone | PRE] Intercepted: {ctx.raw_source.strip()}")

    def on_post(ctx: MorpheHookContext):
        print(f"  [Morphe Standalone | POST] Completed in {ctx.elapsed_ms:.3f}ms")

    kernel.register_hook("pre_run", on_pre)
    kernel.register_hook("post_run", on_post)

    print("=== Standalone Morphe Kernel Demo ===")
    kernel.runsource("alpha = 100 + 337")
    kernel.runsource("print(f'Evaluated: {alpha}')")


if __name__ == "__main__":
    main()

```

#### 8. `bin/morphe-harness`

```bash
#!/usr/bin/env bash
# ==============================================================================
# bin/morphe-harness
# Unshare Multi-Namespace Orchestrator: Bind-Mounts Shims & Spawns Subsystems
# ==============================================================================
set -euo pipefail

# 1. Self-bootstrap into User and Mount Namespaces if unprivileged
if [ "$(id -u)" -ne 0 ]; then
    echo -e "\033[35m[Morphe Sandbox]\033[0m Spawning private User + Mount namespaces (unshare -rm)..."
    exec unshare -rm --fork "$0" "$@"
fi

# We are now UID 0 with CAP_SYS_ADMIN inside the isolated mount namespace
WORKDIR="$(mktemp -d /tmp/morphe_mounts_XXXXXX)"
cleanup() {
    rm -rf "$WORKDIR"
}
trap cleanup EXIT

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"
PYTHONPATH="${SCRIPT_DIR}:${PYTHONPATH:-}"
export PYTHONPATH

echo -e "\033[32m[Morphe Sandbox]\033[0m Initialized in namespace. Root mapping confirmed."

# ------------------------------------------------------------------------------
# 2. Bind-Mount Python Global sitecustomize.py
# ------------------------------------------------------------------------------
PY_SITE="$(python3 -c 'import site; print(site.getsitepackages()[0])' 2>/dev/null || true)"
if [ -n "$PY_SITE" ] && [ -d "$PY_SITE" ]; then
    mkdir -p "$WORKDIR/site_overlay"
    cp -a "$PY_SITE/." "$WORKDIR/site_overlay/" 2>/dev/null || true
    cp "$SCRIPT_DIR/morphe/sitecustomize.py" "$WORKDIR/site_overlay/sitecustomize.py"
    mount --bind "$WORKDIR/site_overlay" "$PY_SITE"
    echo "  ↳ Hooked sitecustomize.py over $PY_SITE"
fi

# ------------------------------------------------------------------------------
# 3. Bind-Mount Dash & Bash Shims
# ------------------------------------------------------------------------------
python3 -c "
from pathlib import Path
from morphe.shell_shims import generate_dash_shim, generate_bashrc_shim
generate_dash_shim(Path('$WORKDIR'))
generate_bashrc_shim(Path('$WORKDIR'))
"

if [ -f "/bin/dash" ] && [ -f "$WORKDIR/dash_shim" ]; then
    mount --bind "$WORKDIR/dash_shim" "/bin/dash"
    echo "  ↳ Bound Morphe proxy shim over /bin/dash"
fi

# ------------------------------------------------------------------------------
# 4. Dispatch Target Command
# ------------------------------------------------------------------------------
if [ "$#" -gt 0 ]; then
    exec "$@"
else
    echo -e "\n=== Running Standalone Test Pipeline ==="
    python3 "$SCRIPT_DIR/morphe/kernel.py"
    
    echo -e "\n=== Spawning Interactive Hooked Bash ==="
    exec /bin/bash --rcfile "$WORKDIR/morphe_bashrc" -i
fi

```

from morphe.context import MorpheHookContext


def test_kernel_lifecycle_hooks():

        events.append(f"PRE:{ctx.raw_source.strip()}")
