import os, pathlib, hashlib, json, sys
from datetime import datetime, timezone
from concurrent.futures import ThreadPoolExecutor, as_completed

DEFAULT_EXTENSIONS = {".toml",".json",".yaml",".yml",".sh",".ts",".py",".rs",".env",".md"}
IGNORE_DIRS = {".git","node_modules",".cache","dist","build","target","__pycache__","vendor",".venv",".venvs",".cargo",".rustup"}

def read_and_hash_file(path, max_kb=200):
    try:
        p = pathlib.Path(path)
        if not p.is_file():
            return None
        if p.stat().st_size > max_kb * 1024:
            return None
        content = p.read_text(encoding="utf-8", errors="replace")
        sha = hashlib.sha256(content.encode()).hexdigest()
        return {"path": str(p), "name": p.name, "size": p.stat().st_size, "sha256": sha, "mtime": p.stat().st_mtime, "content": content[:20000]}
    except Exception as e:
        return {"path": str(path), "error": str(e)}

def collect_sovereign_files(paths, extensions, max_kb=200, max_files=50, num_workers=4):
    target_paths = []
    for root in paths:
        root = os.path.expanduser(root)
        if not os.path.exists(root):
            continue
        for dirpath, dirnames, filenames in os.walk(root):
            dirnames[:] = [d for d in dirnames if d not in IGNORE_DIRS and not d.startswith(".")]
            for fname in filenames:
                ext = os.path.splitext(fname)[1].lower()
                if ext in extensions or fname in ("pi","grok","pitchfork.toml",".env"):
                    target_paths.append(os.path.join(dirpath, fname))
    target_paths = sorted(set(target_paths), key=lambda x: os.path.getmtime(x) if os.path.exists(x) else 0, reverse=True)[:max_files]
    print(f"Scanning {len(target_paths)} files with {num_workers} workers...")
    collected = []
    with ThreadPoolExecutor(max_workers=num_workers) as executor:
        futures = {executor.submit(read_and_hash_file, p, max_kb): p for p in target_paths}
        try:
            for fut in as_completed(futures, timeout=30):
                try:
                    res = fut.result(timeout=5)
                    if res:
                        collected.append(res)
                except Exception as e:
                    print(f"Future failed {e}")
        except KeyboardInterrupt:
            print("Interrupted – shutting down without deadlock")
            executor.shutdown(wait=False, cancel_futures=True)
            sys.exit(130)
    return collected

def main():

def read_and_hash_file(path_str, max_kb=512):
    try:
        p = Path(path_str)
        stat = p.stat(follow_symlinks=False)
        size_kb = stat.st_size / 1024
        is_truncated = False

        if size_kb > max_kb:
            content = f"[TRUNCATED: File size {size_kb:.2f} KB exceeds {max_kb} KB limit]"
            is_truncated = True
        else:
            content = p.read_text(encoding="utf-8", errors="replace")

        sha256 = hashlib.sha256(content.encode("utf-8")).hexdigest()

        return {
            "path": path_str,
            "filename": p.name,
            "extension": p.suffix,
            "size_bytes": stat.st_size,
            "mtime": datetime.fromtimestamp(stat.st_mtime, timezone.utc).isoformat(),
            "is_symlink": p.is_symlink(),
            "is_truncated": is_truncated,
            "sha256": sha256,
            "content": content
        }
    except Exception as err:
        return {"path": path_str, "error": str(err)}


def collect_sovereign_files(targets, extensions, max_kb=512, max_files=None):
    raw_paths, winner_engine, disc_ms = discover_paths_racing(targets)
    now = time.time()

    candidates = []
    seen = set()

    for p_str in raw_paths:
        if p_str in seen:
            continue
        seen.add(p_str)

        fname = os.path.basename(p_str)
        ext = os.path.splitext(fname)[1].lower()

        if ext not in extensions and fname not in EXPLICIT_FILENAMES:
            continue

        try:
            st = os.stat(p_str, follow_symlinks=False)
            is_core = fname in ("pitchfork.toml", "mcp.json", "fork-manifest.json", "pi", "grok")
            is_sov = "sovereign" in p_str.lower()
            base_score = 1000 if is_core else (500 if is_sov else 100)
            score = base_score + (st.st_mtime / now) * 50
            candidates.append((score, st.st_mtime, p_str))
        except Exception:
            pass

    # Sort by priority and timestamp
    candidates.sort(key=lambda x: (x[0], x), reverse=True)
    if max_files is not None and max_files > 0:
        candidates = candidates[:max_files]

    target_paths = [c[2] for c in candidates]

    # Concurrent file ingestion
    t0_read = time.perf_counter()
    num_workers = min(32, (os.cpu_count() or 4) * 4)
    with concurrent.futures.ThreadPoolExecutor(max_workers=num_workers) as executor:
        collected = list(executor.map(lambda p: read_and_hash_file(p, max_kb=max_kb), target_paths))
    read_ms = (time.perf_counter() - t0_read) * 1000

    print(f"⚡ [Discovery Race Winner] : {winner_engine} ({disc_ms:.2f} ms)", file=sys.stderr)
    print(f"⚡ [Concurrent Ingestion] : {len(collected)} files ({read_ms:.2f} ms across {num_workers} threads)", file=sys.stderr)

    return collected


def write_chunks(files, out_dir, chunk_kb=500):
    out_dir = Path(out_dir).expanduser().resolve()
    out_dir.mkdir(parents=True, exist_ok=True)
    max_chunk_bytes = chunk_kb * 1024

    chunks_meta = []
    current_chunk = []
    current_bytes = 0
    part_idx = 1

    def flush_chunk(part_num, items):
        out_name = f"workspace_dump_part_{part_num:04d}.json"
        out_path = out_dir / out_name
        payload = {
            "part": part_num,
            "record_count": len(items),
            "created_at": datetime.now(timezone.utc).isoformat(),
            "files": items
        }
        raw_b = dump_bytes(payload, indent=True)
        out_path.write_bytes(raw_b)

        return {
            "part": part_num,
            "filename": out_name,
            "path": str(out_path),
            "file_count": len(items),
            "size_bytes": len(raw_b),
            "size_kb": round(len(raw_b) / 1024, 2),
            "sha256": hashlib.sha256(raw_b).hexdigest()
        }

    for rec in files:
        rec_encoded = dump_bytes(rec)
        rec_size = len(rec_encoded)

        if current_chunk and (current_bytes + rec_size > max_chunk_bytes):
            meta = flush_chunk(part_idx, current_chunk)
            chunks_meta.append(meta)
            part_idx += 1
            current_chunk = []
            current_bytes = 0

        current_chunk.append(rec)
        current_bytes += rec_size

    if current_chunk:
        meta = flush_chunk(part_idx, current_chunk)
        chunks_meta.append(meta)

    manifest = {
        "generated_at": datetime.now(timezone.utc).isoformat(),
        "total_files": len(files),
        "total_chunks": len(chunks_meta),
        "chunk_size_limit_kb": chunk_kb,
        "chunks": chunks_meta
    }
    manifest_path = out_dir / "workspace_dump_manifest.json"
    manifest_path.write_bytes(dump_bytes(manifest, indent=True))

    print(f"✓ Created {len(chunks_meta)} chunk parts in {out_dir}", file=sys.stderr)
    print(f"✓ Manifest written to: {manifest_path}", file=sys.stderr)
    return manifest_path
